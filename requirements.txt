# Rainette Music — standalone app dependencies.

# Local HTTP + WebSocket server.
aiohttp>=3.9
cryptography>=42.0
zeroconf>=0.131
qrcode[pil]>=8.0
filelock>=3.13

# Search + ad-free audio-stream extraction (YouTube etc.).
# No upper bound on purpose — yt-dlp ships frequent extractor fixes as sites
# change, and pinning it would silently break playback over time.
yt-dlp>=2024.1

# Artist / album / song catalog metadata (optional but recommended;
# the app degrades to yt-dlp-only search without it).
ytmusicapi>=1.12

# Tags and durations for the MP3s already on this computer. Optional in the
# same way yt-dlp is: without it local_library falls back to reading artist and
# title out of the filename. Preferred over porting the phone's hand-rolled
# ID3/MP4 reader because it also understands FLAC, Ogg, Opus, WAV and AIFF, and
# reports real durations rather than zero.
mutagen>=1.47

# Native desktop window (WebView2 on Windows). The autoplay policy patch is
# deliberately bound to this audited pywebview implementation; changing the
# version requires the native playback smoke test to prove the flag still lands.
# The app still falls back to Edge --app mode if pywebview is unavailable.
pywebview==6.2.1

# Only needed to build the shipped app — see release/build-windows-release.ps1.
pyinstaller>=6.0
